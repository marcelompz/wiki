components/ProductCard.tsx
tsx"use client";

import { motion } from "framer-motion";

interface Product {
  id: string;
  name: string;
  tagline: string;
  description: string;
  price: string;
  features: string[];
  cta: string;
  highlight: boolean;
}

export default function ProductCard({
  product,
  index,
}: {
  product: Product;
  index: number;
}) {
  return (
    <motion.div
      initial={{ opacity: 0, y: 40 }}
      whileInView={{ opacity: 1, y: 0 }}
      viewport={{ once: true, margin: "-50px" }}
      transition={{ duration: 0.5, delay: index * 0.15 }}
      whileHover={{ y: -8 }}
      className={`relative rounded-2xl p-8 flex flex-col ${
        product.highlight
          ? "bg-gradient-to-b from-purple-900 to-slate-900 text-white shadow-2xl scale-105 z-10"
          : "bg-white border border-slate-200 shadow-sm"
      }`}
    >
      {product.highlight && (
        <div className="absolute -top-4 left-1/2 -translate-x-1/2 bg-teal-400 text-slate-900 text-xs font-bold px-4 py-1 rounded-full">
          MÁS POPULAR
        </div>
      )}

      <h3 className={`text-2xl font-bold mb-1 ${product.highlight ? "text-white" : "text-purple-900"}`}>
        {product.name}
      </h3>
      <p className={`text-sm mb-4 ${product.highlight ? "text-teal-300" : "text-teal-600"}`}>
        {product.tagline}
      </p>
      <p className={`text-sm mb-6 flex-grow ${product.highlight ? "text-slate-300" : "text-slate-600"}`}>
        {product.description}
      </p>

      <ul className="space-y-2 mb-8">
        {product.features.map((f) => (
          <li key={f} className="flex items-start gap-2 text-sm">
            <span className={product.highlight ? "text-teal-400" : "text-teal-600"}>✓</span>
            <span className={product.highlight ? "text-slate-200" : "text-slate-700"}>{f}</span>
          </li>
        ))}
      </ul>

      <div className="mt-auto">
        <p className={`text-2xl font-bold mb-4 ${product.highlight ? "text-white" : "text-purple-900"}`}>
          {product.price}
        </p>
        <motion.a
          whileHover={{ scale: 1.03 }}
          whileTap={{ scale: 0.98 }}
          href={`#comprar-${product.id}`}
          className={`block w-full text-center font-semibold py-3.5 rounded-xl transition ${
            product.highlight
              ? "bg-teal-400 hover:bg-teal-300 text-slate-900"
              : "bg-purple-900 hover:bg-purple-800 text-white"
          }`}
        >
          {product.cta}
        </motion.a>
      </div>
    </motion.div>
  );
}

2. Guía de Animaciones con Framer Motion para OmniFlow
Instalación
Bashnpm install framer-motion
Principios de animación OmniFlow

























PrincipioAplicaciónVelocidadTransiciones rápidas (0.4–0.6s)FlujoMovimientos suaves de abajo hacia arriba (y)PrecisiónEvitar rebotes exagerados (ease: "easeOut")SutilezaPreferir opacity + y antes que rotaciones fuertes
Animaciones recomendadas
1. Entrada de secciones (whileInView)
tsx<motion.div
  initial={{ opacity: 0, y: 40 }}
  whileInView={{ opacity: 1, y: 0 }}
  viewport={{ once: true, margin: "-100px" }}
  transition={{ duration: 0.6, ease: "easeOut" }}
>
  Contenido
</motion.div>
2. Stagger (aparición escalonada de cards)
tsxconst container = {
  hidden: { opacity: 0 },
  show: {
    opacity: 1,
    transition: { staggerChildren: 0.12 }
  }
};

const item = {

};
3. Hover en botones y cards
tsx<motion.button
  whileHover={{ scale: 1.03, y: -2 }}
  whileTap={{ scale: 0.98 }}
  transition={{ type: "spring", stiffness: 400, damping: 20 }}
>
  Comprar
</motion.button>

tsx<motion.div

    ease: "easeInOut"

Instala Framer Motion
Usa "use client" en los componentes que animan
Prioriza whileInView para secciones
Usa staggerChildren en las 3 cards del Tridente
Añade whileHover + whileTap en todos los botones de compra
Mantén duraciones entre 0.4s y 0.65s
Prefiere easeOut o spring suave

import { useState, useEffect } from "react";

const storySteps = [
  {
    id: 1,
    title: "La visión inicial",
    text: "Nació como un acelerador comercial. La idea era simple y poderosa: darle a empresas con sistemas informáticos anticuados un canal de e-commerce y CRM que se conectara a su sistema preexistente, actualizara ventas e inventario, y fungiera como el frontend de cualquier plataforma.",

  },

    id: 2,
    title: "Escalamiento horizontal",
    text: "Pronto dejó de ser solo un frontend. Empezó a abarcar más canales, más integraciones y más puntos de contacto con el cliente. El flujo ya no era solo de pedidos… era omni-canal.",
    brand: "OrderFlow",
  },
  {
    id: 3,
    title: "Escalamiento vertical",

    id: 4,

    text: "Lo que empezó siendo un acelerador de ventas terminó cargando todo el flujo del negocio sobre sus hombros. OrderFlow se quedó pequeño. Nació OmniFlow.",
    brand: "OmniFlow",
  },
];

export default function EvolutionStory() {
  const [currentStep, setCurrentStep] = useState(0);
  const [isAnimating, setIsAnimating] = useState(false);

  // Auto-play de la historia
  useEffect(() => {
    const interval = setInterval(() => {
      setCurrentStep((prev) => (prev + 1) % storySteps.length);
    }, 5200);
    return () => clearInterval(interval);
  }, []);

  const current = storySteps[currentStep];
  const isOmni = current.brand === "OmniFlow";

  return (
    <section className="relative py-28 overflow-hidden bg-slate-950 text-white">
      {/* Fondo sutil */}
      <div className="absolute inset-0 bg-[radial-gradient(ellipse_at_center,_var(--tw-gradient-stops))] from-purple-900/20 via-slate-950 to-slate-950" />

      <div className="relative max-w-5xl mx-auto px-6">
        {/* Título de sección */}
        <motion.div
          initial={{ opacity: 0, y: 20 }}
          whileInView={{ opacity: 1, y: 0 }}
          viewport={{ once: true }}
          className="text-center mb-16"
        >
          <p className="text-teal-400 text-sm font-semibold tracking-widest mb-3">
            NUESTRA EVOLUCIÓN
          </p>
          <h2 className="text-3xl md:text-5xl font-bold">
            De OrderFlow a OmniFlow
          </h2>
        </motion.div>

        {/* ===== ANIMACIÓN DEL NOMBRE ===== */}
        <div className="flex justify-center mb-16 h-28 md:h-36 items-center">
          <AnimatePresence mode="wait">
            <motion.div
              key={current.brand}
              initial={{ opacity: 0, scale: 0.8, filter: "blur(8px)" }}
              animate={{ opacity: 1, scale: 1, filter: "blur(0px)" }}
              exit={{ opacity: 0, scale: 1.1, filter: "blur(8px)" }}
              transition={{ duration: 0.7, ease: "easeInOut" }}
              className="text-center"
            >
              <div className="flex items-center justify-center gap-3 md:gap-5">
                {/* Isotipo simplificado */}
                <motion.div
                  animate={{
                    rotate: isOmni ? [0, 8, -8, 0] : 0,
                  }}
                  transition={{ duration: 2.5, repeat: isOmni ? Infinity : 0, ease: "easeInOut" }}
                  className="w-14 h-14 md:w-20 md:h-20 rounded-full bg-gradient-to-br from-purple-600 to-teal-400 flex items-center justify-center shadow-lg shadow-teal-500/20"
                >
                  <svg
                    viewBox="0 0 24 24"
                    className="w-7 h-7 md:w-10 md:h-10 text-white"
                    fill="none"
                    stroke="currentColor"
                    strokeWidth="2"
                  >
                    <path d="M13 2L4 14h7l-1 8 10-12h-7l1-8z" />
                  </svg>
                </motion.div>

                {/* Nombre animado */}
                <div className="text-4xl md:text-6xl font-bold tracking-tight">
                  {isOmni ? (
                    <>
                      <span className="text-purple-400">Omni</span>
                      <span className="text-teal-400">Flow</span>
                    </>
                  ) : (
                    <>
                      <span className="text-purple-400">Order</span>
                      <span className="text-teal-400">Flow</span>
                    </>
                  )}
                </div>
              </div>

              <motion.p
                initial={{ opacity: 0 }}
                animate={{ opacity: 1 }}
                className="mt-3 text-sm text-slate-400 tracking-widest"

                {isOmni ? "HIGH-SPEED OMNI-SYSTEM" : "COMMERCIAL ACCELERATOR"}

              <h3 className="text-teal-400 font-semibold mb-4 text-lg">
                {current.title}

                {current.text}

        </div>

Cómo se ve la experiencia

Empieza mostrando OrderFlow + la visión original.
Pasa al escalamiento horizontal (más canales e integraciones).
Luego al escalamiento vertical (se convierte en ERP/CRM completo).
Finalmente ocurre la transformación visual y narrativa a OmniFlow.
El nombre se anima con blur + scale y el isotipo gana un sutil movimiento cuando llega a OmniFlow.
Los usuarios pueden hacer clic en los indicadores para navegar manualmente la historia.
